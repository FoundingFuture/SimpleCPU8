# cycles: the naive microcode set, yours to change.
# One section per instruction, one row of signals per line.

fetch:
  FETCH

NOP:
  PC_INC

HLT:
  HALT

JMP:
  PC_LOAD

JZ:
  PC_INC
  PC_LOAD_Z

JC:
  PC_INC
  PC_LOAD_C

JN:
  PC_INC
  PC_LOAD_N

JV:
  PC_INC
  PC_LOAD_V

JSR:
  PC_INC
  STK_WRITE_PCH
  SP_DEC
  STK_WRITE_PCL
  SP_DEC
  PC_LOAD

JMP D1:
  PC_FROM_D1

JMP D2:
  PC_FROM_D2

JSR D1:
  PC_INC
  STK_WRITE_PCH
  SP_DEC
  STK_WRITE_PCL
  SP_DEC
  PC_FROM_D1

JSR D2:
  PC_INC
  STK_WRITE_PCH
  SP_DEC
  STK_WRITE_PCL
  SP_DEC
  PC_FROM_D2

RET:
  SP_INC
  STK_TO_PCL
  SP_INC
  STK_TO_PCH

  ALU_PASS_B, ACC_LOAD_ALU

  PC_INC

LD A <- [addr8]:

  ALU_PASS_B, FLAGS_LOAD
  PC_INC
